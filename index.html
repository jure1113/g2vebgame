📋 試算表後台 Apps Script 程式碼（複製貼上即可用）
function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName('成績紀錄') || ss.insertSheet('成績紀錄');
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(['時間', '學生代號', '課次編號', '課次名稱', '關卡名稱', '得分', '滿分', '備註', '來源', '紀錄ID']);
  }
  var data = JSON.parse(e.postData.contents);
  var rows = Array.isArray(data) ? data : [data];
  rows.forEach(function(r) {
    sheet.appendRow([
      r.timestamp || new Date().toLocaleString('zh-TW'),
      String(r.studentCode || ''),
      r.lessonId || '',
      r.lessonTitle || '',
      r.stageName || '',
      r.score,
      r.maxScore || 100,
      r.detail || '',
      r.source || '學生闖關',
      r.id || ''
    ]);
  });
  return ContentService.createTextOutput(JSON.stringify({ status: 'success', count: rows.length }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName('成績紀錄');
  if (!sheet || sheet.getLastRow() <= 1) {
    return ContentService.createTextOutput(JSON.stringify({ status: 'success', records: [] }))
      .setMimeType(ContentService.MimeType.JSON);
  }
  var values = sheet.getDataRange().getValues();
  var records = [];
  for (var i = 1; i < values.length; i++) {
    records.push({
      timestamp: String(values[i][0]),
      studentCode: String(values[i][1]),
      lessonId: Number(values[i][2]) || values[i][2],
      lessonTitle: String(values[i][3]),
      stageName: String(values[i][4]),
      score: Number(values[i][5]),
      maxScore: Number(values[i][6]),
      detail: String(values[i][7] || ''),
      source: String(values[i][8] || '雲端紀錄'),
      id: String(values[i][9] || ('cloud_' + i)),
      synced: true
    });
  }
  return ContentService.createTextOutput(JSON.stringify({ status: 'success', records: records }))
    .setMimeType(ContentService.MimeType.JSON);
}
